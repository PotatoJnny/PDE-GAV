cmake_minimum_required(VERSION 3.18)
project(NSPublicInputBridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Nektar++ REQUIRED CONFIG)
find_package(MPI REQUIRED COMPONENTS CXX)
if(NOT "${NEKTAR++_VERSION}" STREQUAL "5.7.0")
  message(FATAL_ERROR "This bridge requires audited Nektar++ 5.7.0")
endif()
add_definitions(${NEKTAR++_DEFINITIONS})
include_directories(${NEKTAR++_INCLUDE_DIRS} ${NEKTAR++_TP_INCLUDE_DIRS})
link_directories(${NEKTAR++_LIBRARY_DIRS} ${NEKTAR++_TP_LIBRARY_DIRS})
add_library(NSPublicInputForce SHARED forcing.cpp)
target_link_libraries(NSPublicInputForce PRIVATE SolverUtils MultiRegions ${NEKTAR++_TP_LIBRARIES})
add_executable(ns_initial_field initial_field.cpp)
target_link_libraries(ns_initial_field PRIVATE MultiRegions ${NEKTAR++_TP_LIBRARIES} MPI::MPI_CXX)

add_executable(ns_evaluate_fields evaluate_fields.cpp)
target_link_libraries(ns_evaluate_fields PRIVATE MultiRegions ${NEKTAR++_TP_LIBRARIES} MPI::MPI_CXX)
